# ---------------------------------------------------------------------------
# Extra pins layered on top of deps/src/vllm-ascend/requirements.txt when the
# AArch64 wheelhouse (deps/python_wheels/) is populated.
#
# Everything here must resolve to a cp310 / manylinux aarch64 wheel, because
# the offline image installs with `pip install --no-index`.
#
# The torch stack matches vllm-ascend v0.13.0, the release upstream builds
# against CANN 8.5.0. See constraints.aarch64.txt and README.aarch64.md.
# ---------------------------------------------------------------------------

# Build backend. setuptools is capped below 81 because vLLM's own metadata
# declares setuptools<81.0.0 and a newer one makes the two installs disagree.
pip
setuptools>=64,<81.0.0
setuptools-scm>=8
wheel
packaging

# jammy ships cmake 3.22.1; vllm-ascend's CMakeLists/pyproject need >= 3.26.
cmake>=3.26
ninja
pybind11

# Torch stack. torch is pinned to the +cpu local-version build from
# download.pytorch.org; constraints.aarch64.txt explains why that matters.
torch==2.8.0+cpu
torch-npu==2.8.0.post2

# vllm-ascend v0.13.0 declares numpy<2.0.0 and CANN's TBE/op tooling is
# validated against the 1.x ABI, so the two agree here.
numpy<2.0.0

# Imported by CANN's TBE/op tooling and by vllm-ascend's setup.py.
attrs
decorator
sympy
cffi
psutil
pyyaml
requests
protobuf
absl-py
googleapis-common-protos
